\newcommand{\ground}[1]{%
    \draw[wire] (#1) -- ++(0,-0.08);
    \draw[wire] ($(#1)+(-0.22,-0.08)$) -- ++(0.44,0);
    \draw[wire] ($(#1)+(-0.15,-0.16)$) -- ++(0.30,0);
    \draw[wire] ($(#1)+(-0.08,-0.24)$) -- ++(0.16,0);
}

\newcommand{\vres}[4]{%
    \draw[wire] (#1,#2) -- (#1,{#2-0.22});
    \draw[wire] ($(#1,#2-0.22)+(-0.27,0)$) rectangle ($(#1,#3+0.22)+(0.27,0)$);
    \draw[wire] (#1,{#3+0.22}) -- (#1,#3);
    \node[label, right=0.30cm] at (#1,{(#2+#3)/2}) {#4};
}

\newcommand{\vind}[4]{%
    \draw[wire] (#1,#2) -- (#1,{#2-0.12});
    \draw[wire] (#1,{#2-0.12})
        .. controls ++(-0.28,-0.08) and ++(-0.28,0.08) .. (#1,{#2-0.35})
        .. controls ++(0.28,-0.08) and ++(0.28,0.08) .. (#1,{#2-0.58})
        .. controls ++(-0.28,-0.08) and ++(-0.28,0.08) .. (#1,{#2-0.81})
        .. controls ++(0.28,-0.08) and ++(0.28,0.08) .. (#1,{#3+0.12});
    \draw[wire] (#1,{#3+0.12}) -- (#1,#3);
    \node[label, right=0.30cm] at (#1,{(#2+#3)/2}) {#4};
}
